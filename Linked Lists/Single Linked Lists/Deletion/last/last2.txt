this is an alternate function which creates two pointers , one stops at second last node , one moves till last 
node , the last node pointer is made free, while second last node pointer ->link is  made NULL

struct node* del_last(struct node *head)
{
    if(head == NULL)
        printf("List is already empty!");
    else if(head->link == NULL)
    {
        free(head);
        head = NULL;
    }
    else
    {
        struct node *temp = head;
        struct node *temp2 = head;
        while(temp->link != NULL)
        {
            temp2 = temp;
            temp = temp->link;
        }
        temp2->link = NULL;
        free(temp);
        temp = NULL;
    }
    return head;
}
